From front/features/chat/model/use-chat.ts — React Query hooks pattern:
```typescript
'use client';

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { getMatches, /* ... */ } from '@/shared/lib/api';

export const chatKeys = {
  matches: ['matches'] as const,
  messages: (chatId: number) => ['chat', chatId, 'messages'] as const,
};

export function useMatchesQuery() {
  return useQuery({
    queryKey: chatKeys.matches,
    queryFn: getMatches,
    staleTime: 30_000,
  });
}

export function useSendMessage(chatId: number) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (content: string) => sendMessage(chatId, content),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: chatKeys.messages(chatId) });
    },
  });
}
```

This plan creates no new trust boundaries. All API calls use the existing apiFetch client which attaches JWT tokens from token-storage.
