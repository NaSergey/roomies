   - `petsOk: boolean` — mix: ~10 true, ~15 false
   - lifestyle scales as `number` (0.00–1.00, two decimal places): `noiseLevel`, `cleanliness`, `sleepSchedule`, `socialLevel`, `workFromHome` — spread across the range, not all the same

3. For each fake user entry, run a transaction:
   ```
   await prisma.$transaction(async (tx) => {
     const user = await tx.user.upsert({
       where: { telegramId: fake.telegramId },
       create: { all fields },
       update: { all mutable fields except telegramId },
     });
     await tx.userDistrict.deleteMany({ where: { userId: user.id } });
     await tx.userDistrict.createMany({ data: [...], skipDuplicates: true });
     await tx.userVibeTag.deleteMany({ where: { userId: user.id } });
     await tx.userVibeTag.createMany({ data: [...], skipDuplicates: true });
     for (const a of fake.quizAnswers) {
       await tx.userQuizAnswer.upsert({ where: { userId_questionId: { userId: user.id, questionId: a.questionId } }, create: {...}, update: {...} });
     }
   });
   ```
